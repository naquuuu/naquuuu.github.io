# Site is intentionally kept out of search results: every page carries a
# robots noindex meta tag. Crawling stays allowed on purpose so search engines
# can see that tag and drop already-indexed URLs. Do not Disallow here until
# Google Search Console shows zero indexed pages, or the noindex is never read.
User-agent: *
Allow: /

# PDFs are not linked from the site anymore. Keep them out of crawlers in case
# an old export gets dropped back in (a PDF cannot carry a noindex meta tag).
Disallow: /assets/private/
Disallow: /portfolio.pdf
